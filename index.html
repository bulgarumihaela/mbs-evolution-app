async function authenticateTelegramUser() { if (!tg) { console.warn( "Telegram WebApp nu este disponibil. Deschide aplicația din Telegram." ); return null; } const initData = tg.initData; if (!initData) { console.warn( "Telegram initData lipsește. Mini App-ul trebuie deschis din Telegram." ); return null; } try { const response = await fetch( TELEGRAM_AUTH_FUNCTION_URL, { method:"POST", headers:{ "Content-Type":"application/json", "apikey":SUPABASE_ANON_KEY }, body:JSON.stringify({ initData:initData }) } ); const result = await response.json(); if (!response.ok || !result.success) { throw new Error( result?.error || "Autentificarea Telegram a eșuat." ); } currentUser = result.user; console.log( "Utilizator MBS autentificat:", currentUser ); if (currentUser?.first_name) { document.getElementById("welcome").textContent = "Bine ai venit, " + currentUser.first_name + " ♡"; document.getElementById("avatarLetter").textContent = currentUser.first_name .charAt(0) .toUpperCase(); } const fullName = [ currentUser?.first_name, currentUser?.last_name ] .filter(Boolean) .join(" "); if (fullName) { document.getElementById("profileName").textContent = fullName; } if (currentUser?.telegram_username) { document.getElementById("profileUsername").textContent = "@" + currentUser.telegram_username; } return currentUser; } catch (error) { console.error( "Eroare autentificare Telegram:", error ); currentUser = null; return null; } }
